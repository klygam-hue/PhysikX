cmake_minimum_required(VERSION 3.18)
project(PhysikX VERSION 0.1.0 LANGUAGES C CXX)
option(PHYSIKX_BUILD_TESTS "Build regression tests" ON)
option(PHYSIKX_BUILD_JAVA "Build optional JNI bridge" OFF)
option(PHYSIKX_SANITIZE "Enable ASan and UBSan" OFF)
add_library(PhysikX SHARED Code/src/physikx.cpp)
add_library(PhysikX::PhysikX ALIAS PhysikX)
target_compile_features(PhysikX PUBLIC cxx_std_17)
target_compile_definitions(PhysikX PRIVATE PX_BUILD)
target_include_directories(PhysikX PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/Code/include> $<INSTALL_INTERFACE:include>)
set_target_properties(PhysikX PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(MSVC)
  target_compile_options(PhysikX PRIVATE /W4 /permissive-)
else()
  target_compile_options(PhysikX PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(PHYSIKX_SANITIZE AND NOT MSVC)
  target_compile_options(PhysikX PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(PhysikX PUBLIC -fsanitize=address,undefined)
endif()
add_executable(physikx_demo Examples/falling_ball.cpp)
target_link_libraries(physikx_demo PRIVATE PhysikX)
add_executable(physikx_benchmark Benchmarks/sparse.cpp)
target_link_libraries(physikx_benchmark PRIVATE PhysikX)
if(PHYSIKX_BUILD_TESTS)
  enable_testing()
  add_executable(physikx_tests Tests/native.cpp)
  target_link_libraries(physikx_tests PRIVATE PhysikX)
  add_test(NAME native COMMAND physikx_tests)
  add_executable(physikx_c_test Tests/c_smoke.c)
  target_link_libraries(physikx_c_test PRIVATE PhysikX)
  add_test(NAME c_abi COMMAND physikx_c_test)
endif()
if(PHYSIKX_BUILD_JAVA)
  find_package(JNI REQUIRED)
  add_library(PhysikXJNI SHARED Bindings/Java/native/bridge.cpp)
  target_include_directories(PhysikXJNI PRIVATE ${JNI_INCLUDE_DIRS})
  target_link_libraries(PhysikXJNI PRIVATE PhysikX)
  set_target_properties(PhysikXJNI PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
include(GNUInstallDirs)
install(TARGETS PhysikX EXPORT PhysikXTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY Code/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT PhysikXTargets NAMESPACE PhysikX:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PhysikX)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/PhysikXConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/PhysikXConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PhysikX)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/PhysikXConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/PhysikXConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/PhysikXConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PhysikX)
